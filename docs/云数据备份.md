## 实测结论（2026-08-13）

- 登录接口 `POST {base}/auth/v1/signin`，请求体 `{"username","password"}`。
  - `username` 用 **`myapp`（昵称）或邮箱**都能登录；**手机号不行**（返回 `INVALID_USERNAME_OR_PASSWORD`）。
  - 返回：`{token_type:"Bearer", access_token, refresh_token, expires_in: 86400, sub}`。**access_token 有效期 24 小时**，所以每天一次的备份任务每次都要先刷新/重登。
- 数据接口需带 `Authorization: Bearer {access_token}`。
- 环境内当前没有任何业务表（`GET /v1/rdb/rest/` 的 OpenAPI 里只有 `guard_system_tables` 两个内置 rpc）。

## 建表 SQL（在云开发控制台 → 数据库 → SQL 执行处跑一次）

```sql
create table if not exists public.app_backup (
  id                bigserial   primary key,
  created_at        timestamptz not null default now(),
  app_version       text        not null,
  db_schema_version integer     not null,
  size_bytes        bigint      not null,
  checksum          text        not null,
  payload           text        not null
);

create index if not exists app_backup_created_at_idx
  on public.app_backup (created_at desc);

comment on table  public.app_backup                   is 'MyAPP 全量加密备份快照（PRD 3.13），一行 = 一次完整备份';
comment on column public.app_backup.app_version       is '产生该备份的 App versionName';
comment on column public.app_backup.db_schema_version is 'Room 数据库 schema 版本，恢复前用于兼容性校验';
comment on column public.app_backup.size_bytes        is '密文字节数（payload Base64 解码后的长度），用于备份列表展示大小';
comment on column public.app_backup.checksum          is '密文的 SHA-256，下载后校验完整性';
comment on column public.app_backup.payload           is 'AES-256-GCM 加密后再 Base64 的全量数据快照；服务端拿不到明文';

-- PostgREST 有 schema 缓存，新建表后必须通知它重新加载，否则接口仍返回 404
notify pgrst, 'reload schema';
```

> `payload` 存的是**密文**：本机用「备份密码」派生密钥做 AES-256-GCM 加密后再 Base64。即使云账号泄露，备份内容仍然打不开。

suspend fun signIn(cloudbase: CloudBaseClient, username: String, password: String): Map<String, Any>? {
    // 账号密码登录
    val result = cloudbase.request<Map<String, Any>>(
        method = "POST",
        path = "/auth/v1/signin",
        body = mapOf("username" to username, "password" to password),
        typeToken = object : TypeToken<Map<String, Any>>() {}
    )

    if (result != null) {
        val accessToken = result["access_token"] as? String
        val refreshToken = result["refresh_token"] as? String
        val userId = result["sub"] as? String

        println("登录成功! 用户ID: $$userId")
        println("访问令牌: $${accessToken?.take(20)}...")

        // 更新访问令牌
        accessToken?.let { cloudbase.updateAccessToken(it) }
        return result
    }
    return null
}

suspend fun getPgData(cloudbase: CloudBaseClient, tableName: String): List<Map<String, Any>>? {
    // 查询 PG 数据库数据
    val data = cloudbase.request<List<Map<String, Any>>>(
        method = "GET",
        path = "/v1/rdb/rest/$tableName?select=*&limit=10",
        typeToken = object : TypeToken<List<Map<String, Any>>>() {}
    )

    if (data != null) {
        println("查询成功: $data")
    }
    return data ?: emptyList()
}

suspend fun addPgData(cloudbase: CloudBaseClient, tableName: String, data: Map<String, Any>): Map<String, Any>? {
    // 新增 PG 数据库数据
    val result = cloudbase.request<Map<String, Any>>(
        method = "POST",
        path = "/v1/rdb/rest/$tableName",
        body = data,
        typeToken = object : TypeToken<Map<String, Any>>() {}
    )

    if (result != null) {
        println("新增成功: $result")
    }
    return result
}

suspend fun updatePgData(cloudbase: CloudBaseClient, tableName: String, dataId: String, data: Map<String, Any>): Any? {
    // 更新 PG 数据库数据
    val result = cloudbase.request<Any>(
        method = "PATCH",
        path = "/v1/rdb/rest/$$tableName?id=eq.$$dataId",
        body = data
    )

    if (result != null) {
        println("更新成功: $result")
    }
    return result
}

suspend fun deletePgData(cloudbase: CloudBaseClient, tableName: String, dataId: String): Boolean {
    // 删除 PG 数据库数据
    val result = cloudbase.request<Any>(
        method = "DELETE",
        path = "/v1/rdb/rest/$$tableName?id=eq.$$dataId"
    )

    if (result != null) {
        println("删除成功")
        return true
    }
    return false
}
